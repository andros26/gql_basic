  addBook(

    ownerId: "6923ab48da41415653382898" 
    
    # 2. Objeto 'input' con los detalles del libro
    input: {
      title: "Cien años de soledad"
      author: "Gabriel García Márquez"
      publicationYear: 1967
      coverImage: "https://ejemplo.com/cien-anios-soledad.jpg" # URL de la imagen (opcional)

  ) {

    author
    ownerId

  }
}


mutation RegistrarResena {
  addReview(
    # 1. ID del usuario que califica
    userId: "6923ab48da41415653382898" 
    
    # 2. Objeto 'input' con los detalles de la reseña
    input: {
      bookId: "6923ac55da41415653382899" 
      rating: 5 # Calificación de 1 a 5
      comment: "Una obra maestra que todo el mundo debería leer. Me encantó la profundidad de los personajes."
    }
  ) {
    # Campos que deseas que el servidor devuelva (tipo Review)
    _id
    bookId
    userId
    rating
    comment
  }
}
